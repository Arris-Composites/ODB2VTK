cmake_minimum_required(VERSION 3.20)
project(ODB2VTK LANGUAGES CXX)

include(CTest)
option(ODB2VTK_BUILD_NATIVE "Build the Abaqus executable and native smoke test" OFF)
set(ODB2VTK_TEST_ODB "" CACHE FILEPATH "Compatible ODB for the native smoke test")

add_library(odb2vtk_core STATIC core/mesh.cpp core/models.cpp core/conversion.cpp
  core/converter.cpp core/in_memory_source.cpp core/writers.cpp core/outputs.cpp
  core/exporter.cpp core/cli.cpp)
target_include_directories(odb2vtk_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(odb2vtk_core PUBLIC cxx_std_17)
set_target_properties(odb2vtk_core PROPERTIES CXX_EXTENSIONS OFF)

if(BUILD_TESTING)
  add_executable(odb2vtk_core_tests tests/test_mesh.cpp)
  target_link_libraries(odb2vtk_core_tests PRIVATE odb2vtk_core)
  add_test(NAME core.mesh COMMAND odb2vtk_core_tests)
  add_executable(odb2vtk_conversion_tests tests/test_conversion.cpp)
  target_link_libraries(odb2vtk_conversion_tests PRIVATE odb2vtk_core)
  add_test(NAME core.conversion COMMAND odb2vtk_conversion_tests)
  add_executable(odb2vtk_writer_tests tests/test_writers.cpp)
  target_link_libraries(odb2vtk_writer_tests PRIVATE odb2vtk_core)
  add_test(NAME core.writers COMMAND odb2vtk_writer_tests)
  add_executable(odb2vtk_cli_tests tests/test_cli.cpp)
  target_link_libraries(odb2vtk_cli_tests PRIVATE odb2vtk_core)
  add_test(NAME core.cli COMMAND odb2vtk_cli_tests)
  add_executable(odb2vtk_writer_fixture tests/writer_fixture.cpp)
  target_link_libraries(odb2vtk_writer_fixture PRIVATE odb2vtk_core)
endif()

if(ODB2VTK_BUILD_NATIVE)
  list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
  find_package(AbqSDK REQUIRED)
  add_library(odb2vtk_abaqus STATIC native/abaqus_source.cpp)
  target_link_libraries(odb2vtk_abaqus PUBLIC odb2vtk_core PRIVATE AbqSDK::Odb)
  add_executable(ODB2VTK main.cpp)
  target_link_libraries(ODB2VTK PRIVATE odb2vtk_abaqus)
  if(BUILD_TESTING)
    add_executable(odb2vtk_export_probe tests/export_probe.cpp)
    target_link_libraries(odb2vtk_export_probe PRIVATE odb2vtk_abaqus)
    add_executable(odb2vtk_source_probe tests/source_probe.cpp)
    target_link_libraries(odb2vtk_source_probe PRIVATE odb2vtk_abaqus)
    add_executable(odb2vtk_source_ownership tests/source_ownership.cpp)
    target_link_libraries(odb2vtk_source_ownership PRIVATE odb2vtk_abaqus)
    add_executable(odb2vtk_double_fixture tests/double_fixture.cpp)
    target_link_libraries(odb2vtk_double_fixture PRIVATE AbqSDK::Odb)
    target_compile_features(odb2vtk_double_fixture PRIVATE cxx_std_17)
    add_executable(odb2vtk_sdk_smoke tests/sdk_smoke.cpp)
    target_link_libraries(odb2vtk_sdk_smoke PRIVATE AbqSDK::Odb)
    target_compile_features(odb2vtk_sdk_smoke PRIVATE cxx_std_17)
    if(ODB2VTK_TEST_ODB)
      if(NOT EXISTS "${ODB2VTK_TEST_ODB}")
        message(FATAL_ERROR "ODB2VTK_TEST_ODB does not exist: ${ODB2VTK_TEST_ODB}")
      endif()
      add_test(NAME native.open_close COMMAND odb2vtk_sdk_smoke "${ODB2VTK_TEST_ODB}")
      set_tests_properties(native.open_close PROPERTIES LABELS "native" TIMEOUT 60)
      add_test(NAME native.source_ownership COMMAND odb2vtk_source_ownership "${ODB2VTK_TEST_ODB}")
      set_tests_properties(native.source_ownership PROPERTIES LABELS "native" TIMEOUT 60)
    endif()
  endif()
endif()
